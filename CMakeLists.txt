cmake_minimum_required(VERSION 3.22)
include(cmake/gcc-arm-none-eabi.cmake)
project(bustepper LANGUAGES C CXX ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
file(GLOB HAL Drivers/STM32G4xx_HAL_Driver/Src/*.c)
list(FILTER HAL EXCLUDE REGEX "(_template|ll_)\\.c$")
file(GLOB_RECURSE CYPHAL Drivers/libcxxcanard/cyphal/*.cpp Drivers/libcxxcanard/libs/*.c)
file(GLOB APP Core/Src/*.c Core/Src/*.cpp)
add_executable(bustepper ${APP} ${HAL} ${CYPHAL} Core/Startup/startup_stm32g474retx.s)
target_include_directories(bustepper PRIVATE Core/Inc Drivers/STM32G4xx_HAL_Driver/Inc Drivers/STM32G4xx_HAL_Driver/Inc/Legacy Drivers/CMSIS/Device/ST/STM32G4xx/Include Drivers/CMSIS/Include Drivers/libcxxcanard Drivers/libcxxcanard/libs)
target_compile_definitions(bustepper PRIVATE USE_HAL_DRIVER STM32G474xx STM32G4 STM32G)
set_property(TARGET bustepper APPEND PROPERTY LINK_DEPENDS "${CMAKE_SOURCE_DIR}/STM32G474XX_FLASH.ld")
target_link_libraries(bustepper PRIVATE m)
add_custom_command(TARGET bustepper POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:bustepper> bustepper.bin COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:bustepper> bustepper.hex)
